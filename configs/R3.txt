! R3 - HQ/edge router: OSPF core, NAT overload, VTY security
hostname R3
!
interface GigabitEthernet0/0
 ip address 10.0.0.3 255.255.255.248
 ip nat inside
 no shutdown
!
interface GigabitEthernet0/1
 ip address 203.0.113.1 255.255.255.252
 ip nat outside
 no shutdown
!
router ospf 1
 router-id 3.3.3.3
 network 10.0.0.0 0.0.0.7 area 0
 default-information originate
!
ip route 0.0.0.0 0.0.0.0 GigabitEthernet0/1
!
! NAT overload (PAT): translate the four internal subnets to the public
! IP on Gi0/1, many-to-one via port mapping
access-list 1 permit 10.10.10.0 0.0.0.255
access-list 1 permit 10.10.20.0 0.0.0.255
access-list 1 permit 10.20.30.0 0.0.0.255
access-list 1 permit 10.20.40.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
!
! Standard ACL restricting VTY (Telnet) management to internal subnets only
access-list 10 permit 10.10.10.0 0.0.0.255
access-list 10 permit 10.10.20.0 0.0.0.255
access-list 10 permit 10.20.30.0 0.0.0.255
access-list 10 permit 10.20.40.0 0.0.0.255
!
line vty 0 4
 access-class 10 in
 password cisco
 login
!
end
